let liveStream = null; async function startLiveBroadcast() { try { // طلب إذن الصوت والصورة مع تفعيل الخصائص الصوتية للآيفون liveStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" }, audio: { echoCancellation: true, noiseSuppression: true } }); const videoElement = document.getElementById('realCameraFeed'); videoElement.srcObject = liveStream; videoElement.muted = true; // كتم الصوت محلياً لمنع التداخل (صدى الصوت)، بينما يُبث للمشاهدين alert('تم تشغيل الكاميرا والبث الصوتي بنجاح!'); } catch (error) { alert('تعذر الوصول للكاميرا أو الميكروفون، يرجى التحقق من إعدادات المتصفح.'); console.error(error); } }
